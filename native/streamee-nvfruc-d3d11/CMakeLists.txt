cmake_minimum_required(VERSION 3.24)
project(streamee_optiflow_d3d11 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This isolated prototype requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(nvfruc_d3d11 STATIC src/pipeline.cpp)
find_program(FXC_EXECUTABLE fxc HINTS
  "C:/Program Files (x86)/Windows Kits/10/bin/10.0.26100.0/x64" REQUIRED)
set(SHADER_HEADERS)
foreach(ENTRY Analysis Cut Validate Repair Warp Down Up Pack Dense Refine)
  set(HEADER "${CMAKE_CURRENT_BINARY_DIR}/shaders/${ENTRY}.h")
  add_custom_command(OUTPUT "${HEADER}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
    COMMAND "${FXC_EXECUTABLE}" /nologo /O3 /Ges /T cs_5_0 /E ${ENTRY}
      /Vn g_${ENTRY} /Fh "${HEADER}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/interpolate.hlsl"
    DEPENDS shaders/interpolate.hlsl VERBATIM)
  list(APPEND SHADER_HEADERS "${HEADER}")
endforeach()
target_sources(nvfruc_d3d11 PRIVATE ${SHADER_HEADERS})
target_include_directories(nvfruc_d3d11 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/shaders")
target_include_directories(nvfruc_d3d11 PUBLIC include)
target_compile_definitions(nvfruc_d3d11 PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(nvfruc_d3d11 PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(nvfruc_d3d11 PUBLIC d3d11 dxgi)
add_library(streamee_optiflow_d3d11 SHARED src/bridge.cpp)
target_compile_definitions(streamee_optiflow_d3d11 PRIVATE STREAMEE_OPTIFLOW_BRIDGE_EXPORTS)
target_compile_options(streamee_optiflow_d3d11 PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(streamee_optiflow_d3d11 PRIVATE nvfruc_d3d11)
add_executable(nvfruc_d3d11_probe tests/probe.cpp)
target_link_libraries(nvfruc_d3d11_probe PRIVATE nvfruc_d3d11 streamee_optiflow_d3d11)
include(CTest)
add_executable(optiflow_pixels tests/pixels.cpp)
target_link_libraries(optiflow_pixels PRIVATE nvfruc_d3d11)
if(BUILD_TESTING)
  add_executable(optiflow_filter_host tests/filter_host.c)
  target_compile_features(optiflow_filter_host PRIVATE c_std_11)
  target_compile_options(optiflow_filter_host PRIVATE /UNDEBUG)
  target_include_directories(optiflow_filter_host PRIVATE include tests)
  add_test(NAME filter_host_no_gpu COMMAND optiflow_filter_host)
  add_executable(nvfruc_d3d11_contracts tests/contracts.cpp)
  target_link_libraries(nvfruc_d3d11_contracts PRIVATE nvfruc_d3d11)
  add_test(NAME d3d11_contracts_no_gpu COMMAND nvfruc_d3d11_contracts)
  add_executable(nvfruc_d3d11_bridge_contracts tests/bridge_contracts.c)
  target_include_directories(nvfruc_d3d11_bridge_contracts PRIVATE include)
  target_link_libraries(nvfruc_d3d11_bridge_contracts PRIVATE streamee_optiflow_d3d11)
  add_test(NAME bridge_contracts_no_gpu COMMAND nvfruc_d3d11_bridge_contracts)
endif()
